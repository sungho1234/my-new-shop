```python
class DataSynchronizer:
    """
    NEW: 모든 지표의 시점 일치 확인
    """
    def __init__(self):
        self.tolerance_ms = 500  # 0.5초 허용

    def validate_sync(self, market_data):
        """
        모든 지표가 동일 시점인지 확인
        """
        current_time = time.time() * 1000  # ms

        # VPIN 업데이트 시간
        vpin_age = current_time - market_data.vpin_last_update

        # 상관관계 업데이트 시간
        corr_age = current_time - market_data.corr_last_update

        # 경고: VPIN이 너무 오래됨 (3분)
        if vpin_age > 180000:
            return False, "VPIN_STALE"

        # 일반적으로 0.5초 이내면 OK
        max_age = max(vpin_age, corr_age)
        if max_age > self.tolerance_ms:
            return False, "DATA_DESYNC"

        return True, "OK"
```

```python
def safe_calculate(func, default_value, error_reason):
    """
    NEW: 안전한 계산 래퍼
    """
    def wrapper(*args, **kwargs):
        try:
            result = func(*args, **kwargs)
            return result, "OK"
        except ZeroDivisionError:
            log_error(f"{func.__name__}: Division by zero")
            return default_value, f"{error_reason}_DIV_ZERO"
        except Exception as e:
            log_error(f"{func.__name__}: {e}")
            return default_value, f"{error_reason}_ERROR"

    return wrapper


# 사용 예시
@safe_calculate(default_value=1.0, error_reason="OB_RATIO")
def calculate_orderbook_imbalance(bids, asks):
    bid_volume = sum(qty for _, qty in bids[:10])
    ask_volume = sum(qty for _, qty in asks[:10])

    if bid_volume + ask_volume < 50:
        return 1.0, False

    ratio = bid_volume / ask_volume  # ZeroDivisionError 가능
    return ratio, True
```

```python
class CorrelationCalculator:
    """
    NEW: 1분마다만 계산 (효율성)
    """
    def __init__(self):
        self.btc_returns = deque(maxlen=60)  # 1시간
        self.eth_returns = deque(maxlen=60)
        self.last_corr = 0.8
        self.last_calc_time = 0
        self.calc_interval = 60  # 1분

    def update_returns(self, btc_price, eth_price):
        """매 1분봉마다 수익률 추가"""
        if len(self.btc_returns) > 0:
            btc_ret = (btc_price - self.btc_last_price) / self.btc_last_price
            eth_ret = (eth_price - self.eth_last_price) / self.eth_last_price

            self.btc_returns.append(btc_ret)
            self.eth_returns.append(eth_ret)

        self.btc_last_price = btc_price
        self.eth_last_price = eth_price

    def get_correlation(self):
        """
        1분마다만 재계산
        """
        now = time.time()

        # 1분 안 지났으면 캐시 반환
        if now - self.last_calc_time < self.calc_interval:
            return self.last_corr

        # 재계산
        if len(self.btc_returns) < 20:
            return 0.8  # 기본값

        try:
            corr = np.corrcoef(
                list(self.btc_returns),
                list(self.eth_returns)
            )[0, 1]

            self.last_corr = corr
            self.last_calc_time = now

            return corr

        except:
            return self.last_corr  # 에러 시 이전 값
```

```python
def analyze_performance_by_time(backtest_results):
    """
    시간대별 성과 비교
    """
    trades = backtest_results.trades

    # 시간대별 분류
    tier1_trades = [t for t in trades if t.time_tier == "TIER1_ASIAN_OPEN"]
    tier2_trades = [t for t in trades if t.time_tier.startswith("TIER2")]
    tier3_trades = [t for t in trades if t.time_tier == "TIER3_US_OPEN"]

    def calc_stats(trades):
        if len(trades) == 0:
            return {"count": 0, "win_rate": 0, "avg_pnl": 0}

        wins = [t for t in trades if t.pnl > 0]
        return {
            "count": len(trades),
            "win_rate": len(wins) / len(trades),
            "avg_pnl": np.mean([t.pnl for t in trades]),
            "total_pnl": sum(t.pnl for t in trades)
        }

    tier1_stats = calc_stats(tier1_trades)
    tier2_stats = calc_stats(tier2_trades)
    tier3_stats = calc_stats(tier3_trades)

    print("=== 시간대별 성과 ===")
    print(f"Tier 1 (09-11시): {tier1_stats['count']}회, "
          f"승률 {tier1_stats['win_rate']:.1%}, "
          f"평균 {tier1_stats['avg_pnl']:.0f}틱")

    print(f"Tier 2 (11-15시, 21-23시): {tier2_stats['count']}회, "
          f"승률 {tier2_stats['win_rate']:.1%}, "
          f"평균 {tier2_stats['avg_pnl']:.0f}틱")

    print(f"Tier 3 (23-01시): {tier3_stats['count']}회, "
          f"승률 {tier3_stats['win_rate']:.1%}, "
          f"평균 {tier3_stats['avg_pnl']:.0f}틱")

    # 결론
    best_tier = max(
        [("TIER1", tier1_stats), ("TIER2", tier2_stats), ("TIER3", tier3_stats)],
        key=lambda x: x[1]['avg_pnl']
    )

    print(f"\n✅ 최고 성과 시간대: {best_tier[0]}")

    # 특정 시간대만 좋으면 집중 권장
    if tier3_stats['avg_pnl'] < 0:
        print("⚠️ Tier 3 (23-01시) 손실 → 비활성화 권장")
```

```python
def analyze_performance_by_regime(backtest_results):
    """
    국면별 성과 (Bull/Bear/Chop 등)
    """
    trades = backtest_results.trades

    regime_groups = {}
    for t in trades:
        regime = t.regime
        if regime not in regime_groups:
            regime_groups[regime] = []
        regime_groups[regime].append(t)

    print("=== 국면별 성과 ===")
    for regime, trades_list in regime_groups.items():
        wins = [t for t in trades_list if t.pnl > 0]
        win_rate = len(wins) / len(trades_list) if trades_list else 0
        avg_pnl = np.mean([t.pnl for t in trades_list]) if trades_list else 0

        print(f"{regime}: {len(trades_list)}회, "
              f"승률 {win_rate:.1%}, "
              f"평균 {avg_pnl:.0f}틱")

    # CHOP에서 손실이면 임계값 상향 권장
    if "CHOP" in regime_groups:
        chop_avg = np.mean([t.pnl for t in regime_groups["CHOP"]])
        if chop_avg < 0:
            print("⚠️ 횡보장(CHOP)에서 손실 → 거래 금지 권장")
```

9.1. 메인 루프
─────────────────────────────────────────────────────────────────

```python
import asyncio
from vpin_calculator import VPINCalculator
from signal_generator import SignalGenerator
from order_manager import OrderManager
from logger import SignalLogger

async def main():
    # 초기화
    vpin_calc = VPINCalculator(bucket_size=40)
    signal_gen = SignalGenerator()
    order_mgr = OrderManager()
    logger = SignalLogger()

    # WebSocket 연결
    ws = await connect_binance_websocket()

    # 메인 루프
    while True:
        # 1. 데이터 수신
        trade = await ws.recv_trade()
        vpin_calc.update(trade)

        orderbook = await ws.recv_orderbook()

        # 2. 1초마다 신호 체크
        if time.time() % 1 < 0.1:  # 대략 1초마다
            market_data = build_market_data(vpin_calc, orderbook)

            # 신호 생성
            signal = signal_gen.check_entry(market_data)

            if signal:
                # 진입
                order = order_mgr.send_entry(signal)
                logger.log_decision("ENTER", market_data, signal.score, "OK")
            else:
                # 거부 (이유 기록)
                logger.log_decision("REJECT", market_data, 0, signal_gen.last_reject_reason)

        # 3. 청산 체크 (매 틱)
        if order_mgr.has_position():
            exit_signal = order_mgr.check_exit(market_data)
            if exit_signal:
                order_mgr.close_position(exit_signal.reason)

        await asyncio.sleep(0.01)


if __name__ == "__main__":
    asyncio.run(main())
```

```python
def run_backtest(data_file, config):
    """
    간단한 백테스팅
    """
    # 데이터 로드
    trades = load_historical_data(data_file)

    vpin_calc = VPINCalculator()
    signal_gen = SignalGenerator()
    portfolio = Portfolio(initial_balance=10000)

    results = []

    for trade in trades:
        # VPIN 업데이트
        vpin_calc.update(trade)

        # 1초마다 신호 체크
        if should_check_signal(trade.time):
            market_data = build_market_data_from_history(trade.time)
            signal = signal_gen.check_entry(market_data)

            if signal:
                # 체결 시뮬레이션
                filled = simulate_fill(signal, config.fill_rate)
                if filled:
                    entry_price = signal.price + config.slippage
                    portfolio.enter_position(entry_price, signal.size)

        # 청산 체크
        if portfolio.has_position():
            exit_signal = check_exit(portfolio.position, market_data)
            if exit_signal:
                exit_price = trade.price - config.slippage
                pnl = portfolio.close_position(exit_price)
                results.append(pnl)

    # 성과 계산
    return calculate_performance(results)
```
